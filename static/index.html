Antigravity Configuration

Add the following configuration to your Antigravity MCP settings file (`mcp_config.json`):

{
  "mcpServers": {
    "orchestrator": {
      "url": "https://orchestrator-d3h8.onrender.com/mcp",
      "headers": {
        "Authorization": "Bearer XRpbN-hS7dZtInpyR4m3LAWri5JrMh4N8J-WUJDmR6M"
      },
      "alwaysAllow": [
        "get_next_task",
        "report_task_progress",
        "create_task",
        "get_project_context",
        "update_project_context",
        "retry_waiting_tasks",
        "list_tasks"
      ]
    }
  }
}

    "headers": {
      "Authorization": "Bearer XRpbN-hS7dZtInpyR4m3LAWri5JrMh4N8J-WUJDmR6M"
    }
  }
}
